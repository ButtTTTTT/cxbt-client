### 4x4变换矩阵
代码出自 Crow_Lz 大佬，感谢大佬！

```
/// <summary>
/// 初始化模型的骨架，计算所有骨骼的绑定姿态矩阵和逆绑定姿态矩阵。
/// 这是进行骨骼动画和蒙皮变换前的必要准备步骤。
/// </summary>
/// <param name="model">目标模型，其Skeleton属性将被赋值。</param>
void InitializeSkeletonPoses(ref Model model)
{
    // 1. 获取骨架引用，确定骨骼数量
    Skeleton skeleton = model.Skeleton; // 假设Skeleton是模型的属性
    int boneCount = skeleton.Bones.Count;
    
    // 2. 初始化矩阵数组
    skeleton.BindPose = new Matrix4x4[boneCount];
    skeleton.InverseBindPose = new Matrix4x4[boneCount];
    
    // 3. 计算绑定姿态矩阵 (Bind Pose)
    // 绑定姿态：每个骨骼从自身局部空间变换到模型空间（或称世界空间）的矩阵
    for (int i = 0; i < boneCount; i++)
    {
        Bone bone = skeleton.Bones[i];
        
        // 3.1 构建骨骼的局部变换矩阵（使用BuildLocalMatrix函数）
        Matrix4x4 localMatrix = BuildLocalMatrix(bone);
        
        // 3.2 根据骨骼层级关系，计算其在绑定姿态下的全局变换
        if (bone.ParentIndex < 0)
        {
            // 根骨骼：没有父级，其全局变换就是局部变换
            skeleton.BindPose[i] = localMatrix;
        }
        else
        {
            // 子骨骼：其全局变换 = 局部变换 × 父骨骼的全局变换
            // 这是关键！通过矩阵连乘，将子骨骼变换到和父骨骼同一的坐标系（模型空间）
            skeleton.BindPose[i] = localMatrix * skeleton.BindPose[bone.ParentIndex];
        }
    }
    
    // 4. 计算逆绑定姿态矩阵 (Inverse Bind Pose)
    // 逆绑定姿态：将顶点从模型空间变换回骨骼自身局部空间的矩阵
    for (int i = 0; i < boneCount; i++)
    {
        // 对每个骨骼的绑定姿态矩阵求逆
        Matrix4x4.Invert(skeleton.BindPose[i], out skeleton.InverseBindPose[i]);
    }
    
    // 5. 将处理好的骨架数据存回模型
    model.Skeleton = skeleton;
}

```
/// <summary>
/// 构建单个骨骼的局部变换矩阵。
/// 这个矩阵描述了骨骼相对于其父骨骼的位置、旋转和缩放变换。
/// </summary>
/// <param name="b">输入的骨骼对象，包含位置(Position)、旋转(Rotation)、缩放(Scale)信息。</param>
/// <returns>返回该骨骼的局部变换矩阵，按缩放(S)、旋转(R)、平移(T)的顺序进行组合。</returns>
static Matrix4x4 BuildLocalMatrix(Bone b)
{
    // 创建平移矩阵：基于骨骼的位置
    var T = Matrix4x4.CreateTranslation(b.Position);
    
    // 创建旋转矩阵：基于骨骼的四元数旋转
    var R = Matrix4x4.CreateFromQuaternion(b.Rotation);
    
    // 创建缩放矩阵：基于骨骼的缩放因子
    var S = Matrix4x4.CreateScale(b.Scale);
    
    // 返回组合后的局部变换矩阵
    // 注意：矩阵乘法顺序为 缩放 -> 旋转 -> 平移
    // 在3D图形学中，这个顺序对应着从局部坐标系变换到父坐标系的完整过程
    return S * R * T;
}
